-- Supermercado: creacion del esquema, tablas e indices (PostgreSQL 15 o superior).
-- Ejecutar conectado a la base de datos donde se quiera crear. Se puede repetir: borra y recrea el esquema tienda.
DROP SCHEMA IF EXISTS tienda CASCADE;
BEGIN;
CREATE SCHEMA tienda;
SET search_path=tienda,pg_catalog;
CREATE TABLE area (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL UNIQUE
);
CREATE TABLE categoria (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL UNIQUE
);
CREATE TABLE empleado (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL,
  documento text NOT NULL UNIQUE,
  direccion text NOT NULL,
  telefono text NOT NULL,
  correo text NOT NULL CHECK(correo ~ '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$'),
  cargo text NOT NULL,
  area_id integer NOT NULL REFERENCES area,
  usuario name NOT NULL UNIQUE
);
CREATE TABLE cliente (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL,
  documento text NOT NULL UNIQUE,
  direccion text NOT NULL,
  telefono text NOT NULL,
  correo text NOT NULL CHECK(correo ~ '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$')
);
CREATE TABLE proveedor (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL,
  nit text NOT NULL UNIQUE,
  direccion text NOT NULL,
  telefono text NOT NULL,
  correo text NOT NULL CHECK(correo ~ '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$'),
  contacto text NOT NULL
);
CREATE TABLE producto (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre text NOT NULL,
  descripcion text NOT NULL,
  categoria_id integer NOT NULL REFERENCES categoria,
  marca text NOT NULL,
  unidad text NOT NULL,
  precio_compra numeric(14,2) NOT NULL CHECK(precio_compra>=0),
  precio_venta numeric(14,2) NOT NULL CHECK(precio_venta>=0),
  minimo numeric(14,3) NOT NULL CHECK(minimo>=0),
  perecedero boolean NOT NULL,
  descontinuado boolean NOT NULL DEFAULT false
);
CREATE TABLE oferta (
  proveedor_id integer NOT NULL REFERENCES proveedor,
  producto_id integer NOT NULL REFERENCES producto,
  costo numeric(14,2) NOT NULL CHECK(costo>=0),
  condiciones text NOT NULL,
  PRIMARY KEY(proveedor_id,producto_id)
);
CREATE TABLE lote (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  producto_id integer NOT NULL REFERENCES producto,
  codigo text NOT NULL,
  ingreso date NOT NULL DEFAULT current_date,
  vence date,
  saldo numeric(14,3) NOT NULL DEFAULT 0 CHECK(saldo>=0),
  UNIQUE(producto_id,codigo),
  CHECK(vence IS NULL OR vence>=ingreso)
);
CREATE TABLE compra (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  fecha timestamptz NOT NULL DEFAULT now(),
  empleado_id integer NOT NULL REFERENCES empleado,
  proveedor_id integer NOT NULL REFERENCES proveedor,
  estado text NOT NULL DEFAULT 'PENDIENTE' CHECK(estado IN ('PENDIENTE','RECIBIDA')),
  fecha_recepcion timestamptz,
  receptor_id integer REFERENCES empleado,
  CHECK((estado='RECIBIDA')=(fecha_recepcion IS NOT NULL) AND (estado='RECIBIDA')=(receptor_id IS NOT NULL))
);
CREATE TABLE venta (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  fecha timestamptz NOT NULL DEFAULT now(),
  empleado_id integer NOT NULL REFERENCES empleado,
  cliente_id integer REFERENCES cliente,
  pago text NOT NULL CHECK(pago IN ('EFECTIVO','TARJETA','TRANSFERENCIA'))
);
CREATE TABLE detalle_compra (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  compra_id integer NOT NULL REFERENCES compra,
  lote_id integer NOT NULL REFERENCES lote,
  cantidad numeric(14,3) NOT NULL CHECK(cantidad>0),
  unitario numeric(14,2) NOT NULL CHECK(unitario>=0),
  subtotal numeric(18,2) GENERATED ALWAYS AS (round(cantidad*unitario,2)) STORED,
  UNIQUE(compra_id,lote_id)
);
CREATE TABLE detalle_venta (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  venta_id integer NOT NULL REFERENCES venta,
  lote_id integer NOT NULL REFERENCES lote,
  cantidad numeric(14,3) NOT NULL CHECK(cantidad>0),
  unitario numeric(14,2) NOT NULL CHECK(unitario>=0),
  subtotal numeric(18,2) GENERATED ALWAYS AS (round(cantidad*unitario,2)) STORED,
  UNIQUE(venta_id,lote_id)
);
CREATE TABLE solicitud (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  fecha timestamptz NOT NULL DEFAULT now(),
  empleado_id integer NOT NULL REFERENCES empleado,
  detalle_id integer NOT NULL REFERENCES detalle_venta,
  tipo text NOT NULL CHECK(tipo IN ('CONSULTA','DEVOLUCION')),
  cantidad numeric(14,3) NOT NULL CHECK(cantidad>0),
  motivo text NOT NULL CHECK(trim(motivo)<>'')
);
CREATE TABLE movimiento (
  id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  fecha timestamptz NOT NULL DEFAULT now(),
  empleado_id integer NOT NULL REFERENCES empleado,
  lote_id integer NOT NULL REFERENCES lote,
  cantidad numeric(14,3) NOT NULL CHECK(cantidad>0),
  tipo text NOT NULL CHECK(tipo IN ('DANO','VENCIMIENTO','PERDIDA','DEVOLUCION_CLIENTE','DEVOLUCION_PROVEEDOR','AJUSTE_ENTRADA','AJUSTE_SALIDA')),
  motivo text NOT NULL CHECK(trim(motivo)<>''),
  solicitud_id integer UNIQUE REFERENCES solicitud,
  CHECK ((tipo='DEVOLUCION_CLIENTE')=(solicitud_id IS NOT NULL))
);

-- Índices de apoyo: las claves foráneas no los crean en PostgreSQL.
CREATE INDEX empleado_area_idx ON empleado(area_id);
CREATE INDEX producto_categoria_idx ON producto(categoria_id);
CREATE INDEX oferta_producto_idx ON oferta(producto_id);
CREATE INDEX lote_vence_idx ON lote(vence) WHERE saldo>0;
CREATE INDEX compra_proveedor_idx ON compra(proveedor_id);
CREATE INDEX compra_empleado_idx ON compra(empleado_id);
CREATE INDEX venta_fecha_idx ON venta(fecha);
CREATE INDEX venta_cliente_idx ON venta(cliente_id) WHERE cliente_id IS NOT NULL;
CREATE INDEX venta_empleado_idx ON venta(empleado_id);
CREATE INDEX detalle_compra_lote_idx ON detalle_compra(lote_id);
CREATE INDEX detalle_venta_lote_idx ON detalle_venta(lote_id);
CREATE INDEX solicitud_detalle_idx ON solicitud(detalle_id);
CREATE INDEX movimiento_lote_idx ON movimiento(lote_id);
COMMIT;
